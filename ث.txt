use ffmpeg_next as ffmpeg;
use anyhow::{Result, Context, anyhow}; // از anyhow برای مدیریت بهتر خطاها استفاده می‌کنیم
use std::env;
use std::path::Path;

fn main() -> Result<()> {
    // ۱. مقداردهی اولیه به FFmpeg
    // این کار باید قبل از هر استفاده دیگری از FFmpeg انجام شود.
    ffmpeg::init().context("خطا در راه‌اندازی FFmpeg. مطمئن شوید که libavformat و بقیه نصب باشند.")?;
    
    println!("\n--- اپلیکیشن app1: تست دیکدینگ فایل TS ---");

    // ۲. دریافت مسیر فایل از آرگومان‌های خط فرمان
    let args: Vec<String> = env::args().collect();
    if args.len() < 2 {
        println!("\n❌ لطفاً مسیر فایل TS را به عنوان آرگومان وارد کنید.");
        println!("   مثال: cargo run -- ./1.ts");
        return Ok(());
    }
    let filepath = &args[1];
    
    // ۳. باز کردن فایل ورودی
    let path = Path::new(filepath);
    let mut ictx = match ffmpeg::format::input(&path) {
        Ok(ictx) => ictx,
        Err(e) => {
            return Err(anyhow!("خطا در باز کردن فایل '{}': {}", filepath, e));
        }
    };
    
    // ۴. پیدا کردن استریم (جریان) ویدیو
    // برای استفاده از متدهای جدید، مستقیماً به StreamContext نیاز داریم.
    let stream_index = ictx.streams().into_iter()
        .position(|s| s.codec().kind() == ffmpeg::media::Type::Video)
        .context("فایل ورودی فاقد جریان ویدیویی است.")?;

    let input_stream = ictx.stream(stream_index)
        .context("خطا در دسترسی به استریم ویدیو.")?;
        
    let video_codec_params = input_stream.parameters();
    
    // ۵. پیدا کردن و باز کردن دیکودر
    // در اینجا از متد id() به جای codec_id() قدیمی استفاده می‌کنیم.
    let video_codec = ffmpeg::codec::find(video_codec_params.id())
        .context(format!("کدک ویدیویی با ID {:?} یافت نشد.", video_codec_params.id()))?;
            
    let mut decoder = video_codec.decoder().open_as(video_codec_params)
        .context("خطا در باز کردن دیکودر.")?;

    println!("✔ اطلاعات فایل TS:");
    println!("  - کدک ویدیو: {}", video_codec.name());
    println!("  - رزولوشن: {}x{}", decoder.width(), decoder.height());
    
    let mut frame_count = 0;
    let mut frame = ffmpeg::frame::Video::empty();
    
    // ۶. حلقه اصلی: خواندن بسته‌ها (Packet) و دیکودینگ آن‌ها به فریم‌ها
    for (stream, packet) in ictx.packets() {
        if stream.index() == stream_index {
            // ارسال بسته به دیکودر
            if let Err(e) = decoder.send_packet(&packet) {
                eprintln!("خطا در ارسال بسته: {}", e);
                continue;
            }
            
            // دریافت فریم‌های دیکد شده
            while decoder.receive_frame(&mut frame).is_ok() {
                frame_count += 1;
                if frame_count <= 5 { // فقط اطلاعات چند فریم اول را نمایش می دهیم
                    println!("  -> فریم ویدیو دیکد شده: #{} (PTS: {})", 
                        frame_count, frame.pts().unwrap_or(0));
                }
            }
        }
        if frame_count >= 10 { // برای جلوگیری از اجرای طولانی، بعد از ۱۰ فریم متوقف می شویم
            break;
        }
    }
    
    // ۷. تخلیه باقیمانده دیکودر (Flush)
    decoder.send_eof()?;
    while decoder.receive_frame(&mut frame).is_ok() {
         frame_count += 1;
    }

    println!("\n✔ دیکدینگ با موفقیت انجام شد. مجموع فریم‌های پردازش شده: {}", frame_count);
    
    Ok(())
}
