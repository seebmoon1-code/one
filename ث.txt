// src-tauri/src/main.rs

#![cfg_attr(not(debug_assertions), windows_subsystem = "windows")]

use tauri::Manager;
use ffmpeg_next as ffmpeg;
use anyhow::{Result, Context, anyhow};
use std::path::Path;

// ساختار برای ارسال اطلاعات فایل به فرانت‌اند
#[derive(serde::Serialize)]
struct VideoInfo {
    codec: String,
    resolution: String,
    duration_secs: f64,
}

// فرمان Tauri برای پردازش فایل TS و استخراج اطلاعات
#[tauri::command]
fn get_ts_file_info(filepath: String) -> Result<VideoInfo, String> {
    // ۱. مقداردهی اولیه FFmpeg
    if let Err(e) = ffmpeg::init() {
        return Err(format!("خطا در راه‌اندازی FFmpeg: {}", e));
    }

    let path = Path::new(&filepath);
    let mut ictx = match ffmpeg::format::input(&path) {
        Ok(ictx) => ictx,
        Err(e) => return Err(format!("خطا در باز کردن فایل: {}", e)),
    };

    // ۲. پیدا کردن استریم ویدیو (استفاده از سازگارترین متد API)
    let stream_index = ictx.streams().into_iter()
        // .id().is_video() سازگارترین متد برای نسخه های مختلف ffmpeg-next
        .position(|s| s.parameters().id().is_video()) 
        .map_err(|_| "فایل فاقد جریان ویدیویی است.".to_string())?;

    let input_stream = ictx.stream(stream_index).map_err(|_| "خطا در دسترسی به استریم.".to_string())?;
    let video_codec_params = input_stream.parameters();

    // ۳. پیدا کردن و باز کردن دیکودر
    let video_codec = match ffmpeg::codec::find(video_codec_params.codec_id()) {
        Some(c) => c,
        None => return Err(format!("کدک ویدیویی با ID {:?} یافت نشد.", video_codec_params.codec_id())),
    };

    let decoder = match video_codec.decoder().open_as(video_codec_params) {
        Ok(d) => d,
        Err(e) => return Err(format!("خطا در باز کردن دیکودر: {}", e)),
    };

    // ۴. محاسبه مدت زمان (duration)
    let duration = if ictx.duration() > 0 { 
        ictx.duration() as f64 / f64::from(ffmpeg::model::time_base::MICROSECONDS)
    } else { 0.0 };

    Ok(VideoInfo {
        codec: video_codec.name().to_string(),
        resolution: format!("{}x{}", decoder.width(), decoder.height()),
        duration_secs: duration,
    })
}


fn main() {
    tauri::Builder::default()
        // ثبت فرمان Rust
        .invoke_handler(tauri::generate_handler![get_ts_file_info]) 
        // تنظیمات پوشه وب (src) در tauri.conf.json تنظیم شده است.
        .run(tauri::generate_context!())
        .expect("خطا در اجرای اپلیکیشن Tauri");
}
